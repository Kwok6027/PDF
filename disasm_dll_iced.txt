#!/usr/bin/env python3
"""
disasm_dll_iced.py - disassemble a native (C/C++) DLL/EXE to x86/x64 assembly
using iced-x86, which is independent of capstone (so it works when capstone
won't install). Pure pip:

    pip install iced-x86

What this gives you: ASSEMBLY, labelled per exported function. It does NOT
produce pseudo-C - for that you need Ghidra/IDA (or angr, which needs capstone).
For .NET assemblies use dnfile+dncil instead; this is for native binaries.

Usage:
    python disasm_dll_iced.py PandoraPricer.dll
    python disasm_dll_iced.py PandoraPricer.dll -o pricer.asm
    python disasm_dll_iced.py PandoraPricer.dll --max-bytes 4096
"""

import argparse
import os
import struct


# --------------------------------------------------------------------------
# Minimal PE parsing: bitness, image base, sections, exported functions
# --------------------------------------------------------------------------
class PE:
    def __init__(self, data):
        self.data = data
        self.sections = []          # dicts: name, vaddr, vsize, rawptr, rawsize, exec
        self.is_pe32_plus = False
        self.image_base = 0
        self._parse()

    def _u16(self, o): return struct.unpack_from("<H", self.data, o)[0]
    def _u32(self, o): return struct.unpack_from("<I", self.data, o)[0]
    def _u64(self, o): return struct.unpack_from("<Q", self.data, o)[0]

    def _cstr(self, o, limit=512):
        end = self.data.find(b"\x00", o, o + limit)
        if end == -1:
            end = min(o + limit, len(self.data))
        return self.data[o:end].decode("ascii", "replace")

    def _parse(self):
        if len(self.data) < 0x40 or self.data[:2] != b"MZ":
            raise ValueError("not a PE file (no MZ signature)")
        e = self._u32(0x3C)
        if e + 26 > len(self.data) or self.data[e:e + 4] != b"PE\x00\x00":
            raise ValueError("not a PE file (no PE signature)")
        coff = e + 4
        num_sections = self._u16(coff + 2)
        size_opt = self._u16(coff + 16)
        opt = coff + 20
        magic = self._u16(opt)
        if magic == 0x20B:
            self.is_pe32_plus = True
            self.image_base = self._u64(opt + 24)
            num_off, dir_off = opt + 108, opt + 112
        elif magic == 0x10B:
            self.is_pe32_plus = False
            self.image_base = self._u32(opt + 28)
            num_off, dir_off = opt + 92, opt + 96
        else:
            raise ValueError("unknown optional header magic 0x%x" % magic)

        num_dirs = self._u32(num_off)
        self.data_dirs = []
        for i in range(num_dirs):
            b = dir_off + i * 8
            if b + 8 > len(self.data):
                break
            self.data_dirs.append((self._u32(b), self._u32(b + 4)))

        sec_off = opt + size_opt
        for i in range(num_sections):
            b = sec_off + i * 40
            if b + 40 > len(self.data):
                break
            chars = self._u32(b + 36)
            self.sections.append({
                "name": self.data[b:b + 8].rstrip(b"\x00").decode("ascii", "replace"),
                "vsize": self._u32(b + 8),
                "vaddr": self._u32(b + 12),
                "rawsize": self._u32(b + 16),
                "rawptr": self._u32(b + 20),
                "exec": bool(chars & 0x20000000),       # IMAGE_SCN_MEM_EXECUTE
            })

    def data_dir(self, i):
        return self.data_dirs[i] if i < len(self.data_dirs) else (0, 0)

    def rva_to_off(self, rva):
        for s in self.sections:
            span = max(s["vsize"], s["rawsize"])
            if s["vaddr"] <= rva < s["vaddr"] + span:
                off = rva - s["vaddr"] + s["rawptr"]
                if 0 <= off < len(self.data):
                    return off
        return None

    def section_of(self, rva):
        for s in self.sections:
            span = max(s["vsize"], s["rawsize"])
            if s["vaddr"] <= rva < s["vaddr"] + span:
                return s
        return None

    def exports(self):
        """Return sorted list of (name, rva) for named exports."""
        rva, _ = self.data_dir(0)
        if not rva:
            return []
        off = self.rva_to_off(rva)
        if off is None:
            return []
        try:
            num_names = self._u32(off + 24)
            names_rva = self._u32(off + 32)
        except struct.error:
            return []
        npt = self.rva_to_off(names_rva)
        if npt is None:
            return []
        # We need the function RVAs too; map name index -> ordinal -> EAT entry.
        funcs_rva = self._u32(off + 28)
        ords_rva = self._u32(off + 36)
        eat = self.rva_to_off(funcs_rva)
        ot = self.rva_to_off(ords_rva)
        out = []
        for i in range(num_names):
            try:
                name_off = self.rva_to_off(self._u32(npt + i * 4))
                name = self._cstr(name_off) if name_off is not None else "export_%d" % i
                func_rva = 0
                if eat is not None and ot is not None:
                    ordinal = self._u16(ot + i * 2)
                    func_rva = self._u32(eat + ordinal * 4)
                out.append((name, func_rva))
            except struct.error:
                break
        out.sort(key=lambda t: t[1])
        return out


# --------------------------------------------------------------------------
# Disassembly via iced-x86
# --------------------------------------------------------------------------
def disassemble(path, outpath, max_bytes):
    try:
        from iced_x86 import Decoder, Formatter, FormatterSyntax, FlowControl
    except ImportError:
        print("iced-x86 not found. Install it (no capstone needed):")
        print("    pip install iced-x86")
        print("(if that wheel also won't download, try: pip install distorm3)")
        return False

    with open(path, "rb") as f:
        data = f.read()
    try:
        pe = PE(data)
    except ValueError as e:
        print("  ", e)
        return False

    bitness = 64 if pe.is_pe32_plus else 32
    fmt = Formatter(FormatterSyntax.INTEL)
    fmt.first_operand_char_index = 10

    exps = pe.exports()
    targets = []
    if exps:
        rvas = [r for _, r in exps]
        for idx, (name, rva) in enumerate(exps):
            if rva == 0 or pe.section_of(rva) is None:
                continue
            # window = up to the next exported function, capped
            nxt = rvas[idx + 1] if idx + 1 < len(rvas) else rva + max_bytes
            size = min(max(nxt - rva, 1), max_bytes)
            targets.append((name, rva, size))
    else:
        # No named exports: linearly disassemble the first executable section.
        sec = next((s for s in pe.sections if s["exec"]), None)
        if sec:
            targets.append(("%s (section)" % sec["name"], sec["vaddr"],
                            min(sec["rawsize"], max_bytes * 8)))

    if not targets:
        print("  no executable code located to disassemble.")
        return False

    n_funcs = 0
    with open(outpath, "w", encoding="utf-8") as out:
        out.write("; x86/x64 disassembly via iced-x86 (no capstone).\n")
        out.write("; %d-bit, image base 0x%x. This is assembly, not pseudo-C.\n\n"
                  % (bitness, pe.image_base))
        for name, rva, size in targets:
            off = pe.rva_to_off(rva)
            if off is None:
                continue
            chunk = data[off:off + size]
            if not chunk:
                continue
            start_addr = pe.image_base + rva
            out.write("; ===== %s  (RVA 0x%x) =====\n" % (name, rva))
            decoder = Decoder(bitness, chunk, ip=start_addr)
            for ins in decoder:
                out.write("%016X  %s\n" % (ins.ip, fmt.format(ins)))
                if ins.flow_control == FlowControl.RETURN:
                    out.write(";   -- ret --\n")
            out.write("\n")
            n_funcs += 1

    print("  OK - disassembled %d block(s) to: %s" % (n_funcs, outpath))
    return True


def main(argv=None):
    p = argparse.ArgumentParser(description="Disassemble a native DLL with iced-x86 (capstone-free).")
    p.add_argument("path", help="native .dll/.exe file")
    p.add_argument("-o", "--out", default=None, help="output .asm path (default: <name>.asm)")
    p.add_argument("--max-bytes", type=int, default=2048,
                   help="max bytes to disassemble per function (default 2048)")
    args = p.parse_args(argv)

    outpath = args.out or (os.path.basename(args.path) + ".asm")
    disassemble(args.path, outpath, args.max_bytes)


if __name__ == "__main__":
    main()
